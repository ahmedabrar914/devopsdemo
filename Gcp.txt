for zone in $(gcloud compute zones list --format="value(name)"); do
  vms=$(gcloud compute instances list --filter="zone:($zone)" --format="value(name)")
  for vm in $vms; do
    disk=$(gcloud compute instances describe "$vm" --zone="$zone" --format="value(disks[0].deviceName)")
    image=$(gcloud compute disks describe "$disk" --zone="$zone" --format="value(sourceImage.basename())")
    if [[ -n "$image" ]]; then
      echo -e "$vm\t$zone\t$image"
    fi
  done
done
