function restart_zotelagent {
  log "INFO" "Adding zotel user to vault/adm group and restarting zotelagent"

  sudo usermod -aG "$VAULT_GROUP" zotel || true
  sudo usermod -aG adm zotel || true

  sudo install -d -m 750 -o "$VAULT_USER" -g "$VAULT_GROUP" /var/log/vault || true

  if [ ! -f /var/log/vault/vault-audit.log ]; then
    sudo install -m 640 -o "$VAULT_USER" -g "$VAULT_GROUP" /dev/null /var/log/vault/vault-audit.log || true
  else
    sudo chown "$VAULT_USER:$VAULT_GROUP" /var/log/vault/vault-audit.log || true
    sudo chmod 640 /var/log/vault/vault-audit.log || true
  fi

  aws s3 cp "$ZOTEL_ROLE_CONFIG" "$ZOTEL_INSTALL_DIR/zotelagent/conf/role-config.yaml" || true
  sudo chmod o+r /var/log/syslog /var/log/kern.log /var/log/auth.log 2>/dev/null || true

  sudo systemctl restart zotelagent || true
}

function configure_vault_server_logging {

  log "INFO" "Configuring Vault journal logging"

  cat >/etc/rsyslog.d/10-vault-journal.conf <<EOF
module(load="imjournal" StateFile="imjournal-vault.state")

if (\$programname == "vault") then {
    action(type="omfile" file="/var/log/vault/vault-server.log")
    stop
}
EOF

  touch /var/log/vault/vault-server.log

  chown syslog:adm /var/log/vault
  chmod 775 /var/log/vault

  chown syslog:adm /var/log/vault/vault-server.log
  chmod 664 /var/log/vault/vault-server.log

  systemctl restart rsyslog
}

function configure_vault_server_logging {
  log "INFO" "Configuring Vault journal logging"

  sudo install -d -m 775 -o syslog -g adm /var/log/vault
  sudo install -m 664 -o syslog -g adm /dev/null /var/log/vault/vault-server.log

  sudo tee /etc/rsyslog.d/10-vault-journal.conf >/dev/null <<'EOF'
module(load="imjournal" StateFile="imjournal-vault.state")

if ($syslogtag startswith "vault[") then {
    action(type="omfile" file="/var/log/vault/vault-server.log")
    stop
}
EOF

  sudo rsyslogd -N1
  sudo systemctl restart rsyslog
}
