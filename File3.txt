function configure_zotel_system_logs {
  log "INFO" "Patching Zotel role-config.yaml for system logs"

  ROLE_CONFIG="$ZOTEL_INSTALL_DIR/zotelagent/conf/role-config.yaml"

  sudo cp "$ROLE_CONFIG" "$ROLE_CONFIG.bak" || true

  sudo python3 - <<'PY'
from pathlib import Path

path = Path("/sc/zotelagent/conf/role-config.yaml")
text = path.read_text()

receiver = """\
  filelog/system_logs:
    include:
      - /var/log/syslog
      - /var/log/kern.log
      - /var/log/auth.log
    start_at: end
    include_file_path: true
    include_file_name: true
    operators:
      - type: copy
        from: attributes['log.file.name']
        to: resource['service.name']

"""

if "filelog/system_logs:" not in text:
    text = text.replace("\nservice:\n", "\n" + receiver + "service:\n")

text = text.replace(
    "receivers: [filelog/vault_as_a_service_1]",
    "receivers: [filelog/vault_as_a_service_1, filelog/system_logs]"
)

path.write_text(text)
PY

  sudo usermod -aG adm zotel || true
}
